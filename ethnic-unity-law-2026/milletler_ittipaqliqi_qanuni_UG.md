# جۇڭخۇا خەلق جۇمھۇرىيىتىنىڭ مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتىنى ئىلگىرى سۈرۈش قانۇنى

*(2026-يىلى 3-ئاينىڭ 12-كۈنى 14-نۆۋەتلىك مەملىكەتلىك خەلق ۋەكىللەر قۇرۇلتىيىنىڭ تۆتىنچى قېتىملىق يىغىنىدا تەستىقلاندى)*

## كىرىش سۆز

جۇڭگو دۇنيادىكى تارىخى ئەڭ ئۇزۇن دۆلەتلەرنىڭ بىرى، جۇڭخۇا مىللىتى بەش مىڭ يىلدىن ئارتۇق مەدەنىيەت تارىخىغا ئىگە ئۇلۇغ مىللەت. جۇڭگودىكى ھەر قايسى مىللەتلەر ئۇزاق مۇددەتلىك ئالاقە، ئالماشتۇرۇش، ئارىلىشىش، بىرلىشىش داۋامىدا، ۋەتەننىڭ كەڭ زېمىنىنى بىرلىكتە ئاچتى، بىرلىككە كەلگەن كۆپ مىللەتلىك دۆلەتنى بىرلىكتە قۇردى، شانلىق جۇڭگو تارىخىنى بىرلىكتە يازدى، پارلاق جۇڭخۇا مەدەنىيىتىنى بىرلىكتە ياراتتى، ئۇلۇغ مىللەت روھىنى بىرلىكتە يېتىلدۈردى، قان قېرىنداشلىق، ئوخشاش ئېتىقاد، ئوخشاش مەدەنىيەت، ئۆزئارا بېقىنىدىغان ئىقتىساد، ئۆزئارا يېقىن ھېسسىياتقا ئىگە تەقدىر ئورتاق گەۋدىسىگە ئۇيۇشتى. 

1840-يىلدىن كېيىن، جۇڭگو تەدرىجىي ھالدا يېرىم مۇستەملىكە، يېرىم فېئوداللىق جەمئىيەتكە ئايلاندى. جۇڭگودىكى ھەر قايسى مىللەتلەر دۆلەت زېمىنىنى بۆلۈشكە بولمايدۇ، دۆلەتنى قالايمىقان قىلىشقا بولمايدۇ، مىللەتنى پارچىلاشقا بولمايدۇ، مەدەنىيەتنى ئۈزۈشكە بولمايدۇ دېگەن بىرلىككە كەلگەن بۈيۈك ئېتىقادتا باشتىن -ئاخىر چىڭ تۇرۇپ، دۆلەتنى قۇتقۇزۇش، ساقلاپ قېلىش، تاشقى تاجاۋۇزچىلىققا بىرلىكتە قارشى تۇرۇش يولىدىكى قەھرىمانلىق كۈرەش داۋامىدا، جۇڭخۇا مىللىتىنىڭ ئۆزلۈكىدىن مەۋجۇت بولۇشتىن ئۆزلۈكىدىن ئاڭلىق بولۇشقا قاراپ يۈزلىنىشتەك ئۇلۇغ ئۆزگىرىشنى ئىشقا ئاشۇردى. 

جۇڭگو كوممۇنىستىك پارتىيەسى جۇڭگو ئىشچىلار سىنىپىنىڭ ئالدىنقى سەپ قوشۇنى، شۇنداقلا جۇڭگو خەلقى ۋە جۇڭخۇا مىللىتىنىڭ ئالدىنقى سەپ قوشۇنى بولۇپ، جۇڭگو خەلقىگە بەخت يارىتىش، جۇڭخۇا مىللىتىنىڭ گۈللىنىشىنى ئىشقا ئاشۇرۇشنى باشتىن-ئاخىر ئەسلىي مەقسەت ۋە بۇرچ قىلغان، پۈتۈن مەملىكەتتىكى ھەر قايسى مىللەت خەلقىنى ئىتتىپاقلاشتۇرۇپ يېتەكلەپ، مىللىي مۇستەقىللىق ۋە خەلق ئازادلىقىنى ئىشقا ئاشۇردى، جۇڭخۇا خەلق جۇمھۇرىيىتىنى قۇردى، ھەر قايسى مىللەتلەرنىڭ ھەقىقىي تۈردە باراۋەر سىياسىي ھوقۇققا ئېرىشىشىگە، بىرلىكتە ئۆز ئىشىغا ئۆزى ئىگە بولۇشىغا كاپالەتلىك قىلدى، ھەر قايسى مىللەتلەرنىڭ بىرلىكتە ئىتتىپاقلىشىپ كۈرەش قىلىشىنى، بىرلىكتە گۈللىنىپ تەرەققىي قىلىشىنى ئىشقا ئاشۇردى، جۇڭگوچە مىللەت مەسىلىسىنى ھەل قىلىشنىڭ توغرا يولىنى ئىجادچانلىق بىلەن تېپىپ چىقتى، جۇڭخۇا مىللىتىنىڭ قىياپىتىدە تارىخىي ئۆزگىرىش يۈز بەردى. 

جۇڭگوچە سوتسىيالىزم يېڭى دەۋرگە كىرگەندىن بۇيان، جۇڭگو كوممۇنىستىك پارتىيەسى ماركسىزم مىللەت نەزەرىيەسىنى جۇڭگونىڭ مىللەتلەر مەسىلىسىنىڭ كونكرېت ئەمەلىيىتى بىلەن بىرلەشتۈرۈشنى، جۇڭخۇانىڭ مۇنەۋۋەر ئەنئەنىۋى مەدەنىيىتى بىلەن بىرلەشتۈرۈشنى، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەشنى پارتىيەنىڭ مىللەتلەر خىزمىتىدىكى ئاساسىي لىنىيە، مىللىي رايونلاردىكى ھەر قايسى خىزمەتلەرنىڭ ئاساسىي لىنىيەسى قىلىشتا باشتىن-ئاخىر چىڭ تۇرۇپ، جۇڭگو كوممۇنىستىك پارتىيەسىنىڭ مىللەتلەر خىزمىتىنى كۈچەيتىش ۋە ياخشىلاش توغرىسىدىكى مۇھىم ئىدىيەسىنى شەكىللەندۈردى، ماركسىزم مىللەت نەزەرىيەسىنى جۇڭگوچىلاشتۇرۇش ۋە دەۋر بىلەن تەڭ ئىلگىرى سۈرۈشنىڭ يېڭى سەۋىيەسىنى ئاچتى، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى قۇرۇلۇشىدا تارىخىي نەتىجىلەر قولغا كەلتۈرۈلدى. 

جۇڭخۇا مىللىتى ھەر قايسى مىللەتلەرنىڭ ئۇيۇشۇشىدىن شەكىللەنگەن كۆپ خىل بىر گەۋدە چوڭ ئائىلە. مىللەتلەر ئىتتىپاقلىقى دۆلىتىمىزدىكى ھەر قايسى مىللەت خەلقىنىڭ ھاياتلىق لىنىيەسى. جۇڭخۇا مىللىتىنىڭ ئۇلۇغ گۈللىنىشىنى ئىشقا ئاشۇرۇش پۈتۈن جۇڭخۇا پەرزەنتلىرىنىڭ ئورتاق ئارزۇسى، دۆلەتنىڭ بىرلىكىنى قوغداش، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتىنى ئىلگىرى سۈرۈش پۈتۈن جۇڭگو خەلقىنىڭ ئورتاق مەسئۇلىيىتى. 

پۈتۈن مەملىكەتتىكى ھەر قايسى مىللەت خەلقى، بارلىق دۆلەت ئورگانلىرى ۋە قوراللىق كۈچلەر، ھەر قايسى سىياسىي پارتىيەلەر ۋە ھەر قايسى ئىجتىمائىي تەشكىلاتلار، ھەر قايسى كارخانا، كەسپىي ئورۇنلار جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى قۇرۇلۇشىنى ئىلگىرى سۈرۈشنى ئورتاق ۋەزىپە قىلىشى، جۇڭخۇا مىللىتىنىڭ تۈپ مەنپەئەتى ۋە ئومۇمىي مەنپەئەتىنى نەزەردە تۇتۇشى، ئورتاقلىقنى كۈچەيتىشنى يۆنىلىش قىلىشى، باراۋەر، ئىتتىپاق، ئۆزئارا ياردەم بېرىش، ئىناق سوتسىيالىستىك مىللەت مۇناسىۋىتىنى مۇس تەھكەملىشى ۋە راۋاجلاندۇرۇشى، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىيات ئىشلىرىنى يۇقىرى سۈپەتتە ئىلگىرى سۈرۈشى، جۇڭخۇا مىلل ەتلىرىنىڭ بۈيۈك ئىتتىپاقلىقى ئارقىلىق جۇڭگوچە زامانىۋىلاشتۇرۇشنى ئىلگىرى سۈرۈشى، سوتسىيالىستىك زامانىۋىلاشقان دۆلەتنى ئومۇميۈزلۈك قۇرۇش، جۇڭخۇا مىللىتىنىڭ ئۇلۇغ گۈللىنىشىنى ئومۇميۈزلۈك ئىلگىرى سۈرۈش ئۈچۈن ئىتتىپاقلىشىپ كۈرەش قىلىشى كېرەك.

## 1-باب ئومۇمىي پىرىنسىپ

**1-ماددا** مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتىنى ئىلگىرى سۈرۈش، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى قۇرۇلۇشىنى ئىلگىرى سۈرۈش، جۇڭخۇا مىللىتىنىڭ ئۇلۇغ گۈللىنىشىنى ئىشقا ئاشۇرۇش ئۈچۈن، ئاساسىي قانۇنغا ئاساسەن، بۇ قانۇن تۈزۈلدى.

**2-ماددا** مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىيات ئىشلىرى جۇڭگو كوممۇنىستىك پارتىيەسىنىڭ ئومۇميۈزلۈك رەھبەرلىكىدە چىڭ تۇرىدۇ، جۇڭگوچە سوتسىيالىزمنىڭ ئۇلۇغ بايرىقىنى ئېگىز كۆتۈرىدۇ، ماركسىزم-لېنىنىزم، ماۋ زېدۇڭ ئىدىيەسى، دېڭ شياۋپىڭ نەزەرىيەسى،ۇھىم ئىدىيەسى، ئىلمىي تەرەققىيات قارىشىدا چىڭ تۇرىدۇ، شى جىنپىڭ يېڭى دەۋر جۇڭگوچە سوتسىيالىزم ئىدىيەسىنى ئومۇميۈزلۈك ئىزچىللاشتۇرىدۇ، ھەر قايسى مىللەتلەرنىڭ ئىتتىپاقلىشىپ كۈرەش قىلىشىنىڭ ئورتاق ئىدىيەۋى سىياسىي ئاساسىنى مۇستەھكەملەيدۇ، قۇدرەتلىك دۆلەت قۇرۇش، مىللەتنى گۈللەندۈرۈش ئۈچۈن كۈچ توپلايدۇ.

**3-ماددا** جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەشتە، ھەر قايسى مىللەت ئاممىسىنى تەقدىرداشلىق، ئورتاق شەرەپ-نومۇس، ئورتاق ھايات-مامات، ئورتاق تەقدىر ئورتاق گەۋدىسى ئىدىيەسىنى مۇستەھكەم تۇرغۇزۇشقا يېتەكلەش، جۇڭخۇا مىللىتىنىڭ ئۇيۇشۇش كۈچىنى ئاشۇرۇش كېرەك.

**4-ماددا** جۇڭخۇا مىللىتى ئورتاق گەۋدىسى قۇرۇلۇشىنى ئىلگىرى سۈرۈشتە، ئىقتىساد، سىياسەت، مەدەنىيەت، جەمئىيەت ۋە ئېكولوگىيە مەدەنىيىتى قۇرۇلۇشىنى بىر تۇتاش پىلانلاش، ھەر قايسى مىللەتلەرنىڭ بىرلىكتە گۈللىنىپ تەرەققىي قىلىشىنى ئومۇميۈزلۈك ئىشقا ئاشۇرۇش، ھەر قايسى مىللەت خەلقىنىڭ بىرلىكتە ئۆز ئىشىغا ئۆزى ئىگە بولۇشىغا كاپالەتلىك قىلىش، جۇڭخۇا مىللىتىنىڭ ئورتاق روھىي ماكانىنى بەرپا قىلىش، ھەر قايسى مىللەتلەرنىڭ ئومۇميۈزلۈك ئۆزئارا سىڭىشىشىنى ۋە كەڭ كۆلەمدە ئالاقە، ئالماشتۇرۇش، ئارىلىشىش، بىرلىشىشىنى ئىلگىرى سۈرۈش، ئىنسان بىلەن تەبىئەتنىڭ ئىناق بىرگە مەۋجۇت بولۇپ تۇرۇشىدىكى ئېكولوگىيەلىك ماكانىنى بىرلىكتە قوغداش، جۇڭخۇا مىللىتىنىڭ تونۇش دەرىجىسى تېخىمۇ يۇقىرى، ئۇيۇشۇش كۈچى تېخىمۇ كۈچلۈك تەقدىر ئورتاق گەۋدىسىگە ئايلىنىشىنى ئىلگىرى سۈرۈش كېرەك.

**5-ماددا** جۇڭخۇا خەلق جۇمھۇرىيىتىنىڭ پۇقرالىرى قانۇن ئالدىدا بىردەك باراۋەر، جۇڭخۇا خەلق جۇمھۇرىيىتىدىكى ھەر قايسى مىللەتلەر بىردەك باراۋەر. ھەر قانداق مىللەتنى كەمسىتىش ۋە بېسىمغا ئۇچرىتىش مەنئى قىلىنىدۇ.

**6-ماددا** جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭى مىللەتلەر ئىتتىپاقلىقىنىڭ ئاساسى. دۆلەت ئورتاقلىقنى ئاشۇرۇش، پەرقلىقنى ھۆرمەتلەش ۋە سىغدۇرۇشتا چىڭ تۇرۇپ، ھەر مىللەتنىڭ ئۆزئارا ياردەم بېرىشى، ئىناق ئۆتۈشىنى ئىلگىرى سۈرۈپ، جۇڭخۇا مىللەتلىرىنىڭ بۈيۈك ئىتتىپاقلىقىنى قوغدايدۇ. مىللەتلەر ئىتتىپاقلىقىنى بۇزغۇچى ۋە مىللىي بۆلگۈنچىلىك پەيدا قىلغۇچى ھەرىكەتلەرنى چەكلەيدۇ.

**7-ماددا** دۆلەت ھەر مىللەتنىڭ ئورتاق ئىتتىپاقلىشىپ كۈرەش قىلىشىنى، ئورتاق گۈللىنىپ تەرەققىي قىلىشىنى ئىلگىرى سۈرىدۇ، ماددىي مەدەنىيەت، سىياسىي مەدەنىيەت، مەنىۋى مەدەنىيەت، ئىجتىمائىي مەدەنىيەت، ئېكولوگىيەلىك مەدەنىيەتنىڭ ماس تەرەققىي قىلىشىنى ئىلگىرى سۈرىدۇ، جۇڭخۇا مىللىتىنىڭ تەرەققىياتى ۋە ئىلگىرىلىشىنى ئومۇميۈزلۈك ئىلگىرى سۈرىدۇ.

**8-ماددا** دۆلەت مىللىي تېررىتورىيەلىك ئاپتونومىيە تۈزۈمىدە چىڭ تۇرىدۇ ۋە ئۇنى مۇكەممەللەشتۈرىدۇ، دۆلەتنىڭ بىرلىكى ۋە مىللەتلەر ئىتتىپاقلىقىنى قوغدايدۇ.

**9-ماددا** دۆلەت مىللىي ئىشلارنى قانۇن بويىچە ئىدارە قىلىشتا چىڭ تۇرىدۇ، ھەر مىللەت ئاممىسىنىڭ قانۇنلۇق ھوقۇق -مەنپەئەتىنى قانۇن بويىچە كاپالەتلەندۈرىدۇ، ھەر مىللەت ئاممىسىنىڭ دۆلەت ئېڭى، پۇقرالىق ئېڭى، قانۇنچىلىق ئېڭىنى كۈچەيتىدۇ، مىللىي ئىشلارنى ئىدارە قىلىش سىستېمىسى ۋە ئىدارە قىلىش ئىقتىدارىنى قانۇنچىلىق يولىدا زامانىۋىلاشتۇرىدۇ.

**10-ماددا** جۇڭخۇا خەلق جۇمھۇرىيىتىنىڭ پۇقرالىرى دۆلەتنىڭ بىرلىكى ۋە پۈتۈن مەملىكەتتىكى ھەر مىللەتنىڭ ئىتتىپاقلىقىنى قوغداش مەجبۇرىيىتىگە ئىگە، دۆلەتنىڭ ئىگىلىك ھوقۇقى، بىخەتەرلىكى، تەرەققىيات مەنپەئەتىنى قوغدىشى كېرەك. 

مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىيات ئىشلىرىغا تاشقى كۈچلەرنىڭ ئارىلىشىشىغا يول قويۇلمايدۇ. جۇڭخۇا خەلق جۇمھۇرىيىتىگە مىللەت، دىن، كىشىلىك ھوقۇق قاتارلىق نامدا سىڭىپ كىرىش، بۇزغۇنچىلىق قىلىش، تۆھمەت چاپلاش، قارا چاپلاش، توسۇش، بېسىم قىلىش قىلمىشلىرىغا قەتئىي قارشى تۇرۇلىدۇ.

## 2-باب ئورتاق مەنىۋى ماكان بەرپا قىلىش

**11-ماددا** دۆلەت سوتسىيالىستىك يادرولۇق قىممەت قارىشىنى يېتەكچى قىلىشتا چىڭ تۇرىدۇ، ۋەتەنپەرۋەرلىك، كوللېكتىپچىلىق، سوتسىيالىزم تەربىيەسىنى چوڭقۇرلاشتۇرىدۇ، ھەر مىللەت ئاممىسىنى ۋەتەنپەرۋەرلىكنى يادرو قىلغان مىللىي روھ ۋە ئىسلاھات، يېڭىلىق يارىتىشنى يادرو قىلغان دەۋر روھىنى ئەۋج ئالدۇرۇشقا يېتەكلەيدۇ، بۈيۈك ۋەتەن، جۇڭخۇا مىللىتى، جۇڭخۇا مەدەنىيىتى، جۇڭگو كوممۇنىستىك پارتىيەسى، جۇڭگوچە سوتسىيالىزمغا بولغان ئېتىراپ قىلىشنى مۇستەھكەملەيدۇ.

**12-ماددا** دۆلەت جۇڭگو كوممۇنىستىك پارتىيەسى تارىخى، يېڭى جۇڭگو تارىخى، ئىسلاھات، ئېچىۋېتىش تارىخى، سوتسىيالىزم تەرەققىيات تارىخى تەربىيەسىنى تەشكىللەپ قانات يايدۇرىدۇ، جۇڭخۇا مىللىتى تەرەققىيات تارىخى تەشۋىقات-تەربىيەسىنى چوڭقۇر قانات يايدۇرىدۇ، ھەر مىللەت ئاممىسىنى توغرا دۆلەت قارىشى، تارىخ قارىشى، مىللەت قارىشى، مەدەنىيەت قارىشى، دىن قارىشىنى پۇختا تۇرغۇزۇشقا يېتەكلەيدۇ.

**13-ماددا** دۆلەت سوتسىيالىستىك ئىلغار مەدەنىيەتنى تەرەققىي قىلدۇرىدۇ، ئىنقىلابىي مەدەنىيەتنى ئەۋج ئالدۇرىدۇ، جۇڭخۇا مۇنەۋۋەر ئەنئەنىۋى مەدەنىيىتىنى داۋاملاشتۇرىدۇ، ھەر مىللەت ئاممىسىنى جۇڭخۇا مەدەنىيىتىنى ئېتىراپ قىلىشنى كۈچەيتىش، جۇڭخۇا مەدەنىيىتىگە بولغان ئىشەنچنى ئاشۇرۇشقا يېتەكلەيدۇ. 

ھەر مىللەتنىڭ مۇنەۋۋەر ئەنئەنىۋى مەدەنىيىتى جۇڭخۇا مەدەنىيىتىنىڭ تەركىبىي قىسمىدۇر. دۆلەت سوتسىيالىستىك ئىلغار مەدەنىيەت ئارقىلىق ھەر مىللەتنىڭ مۇنەۋۋەر ئەنئەنىۋى مەدەنىيىتىنىڭ ئىجادچانلىق بىلەن ئۆزگىرىشى ۋە يېڭىلىق يارىتىشچانلىق بىلەن تەرەققىي قىلىشىغا يېتەكچىلىك قىلىشتا چىڭ تۇرىدۇ، جۇڭخۇا مۇنەۋۋەر ئەنئەنىۋى مەدەنىيەت نەتىجىلىرىنى تەشۋىق قىلىش ۋە ئومۇملاشتۇرۇشنى قوللايدۇ. 

ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى جۇڭخۇا مىللىتى ئورتاق گەۋدىسىنىڭ شەكىللىنىشى ۋە تەرەققىياتىنى ئەكس ئەتتۈرىدىغان مەدەنىيەت يادىكارلىقلىرى، ئەنئەنىۋى بىناكارلىق، داڭلىق شەھەر، داڭلىق بازار، داڭلىق كەنت، تارىخىي كوچا رايونى، ئەنئەنىۋى كەنت ۋە غەيرىي ماددىي مەد ەنىيەت مىراسلىرى قاتارلىق ھەر خىل مەدەنىيەت بايلىقلىرىنى قېزىش، قوغداش ۋە مۇۋاپىق ئىشلىتىشنى كۈچەيتىشى كېرەك.

**14-ماددا** دۆلەت ھەر مىللەت ئورتاق ئورتاق بەھرىلىنىدىغان جۇڭخۇا مەدەنىيەت بەلگىسى ۋە جۇڭخۇا مىللىتى ئوبرازىنى تۇرغۇزىدۇ ۋە گەۋدىلەندۈرىدۇ، جۇڭخۇا مىللىتىنىڭ مەدەنىيەت ۋە تەبىئىي مىراسلىرى قاتارلىق بايلىقلارغا تايىنىپ، جۇڭخۇا مەدەنىيەت بەلگىسى سىستېمىسىنىڭ قۇرۇلۇشىنى ئىلگىرى سۈرىدۇ. 

ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى ئاممىۋى ئەسلىھە، بىناكارلىق لايىھەسى، مەنزىرە رايونى كۆرگەزمىسى، يەر نامى ئاتاش ۋە ئاممىۋى پائالىيەتلەر جەھەتتە جۇڭخۇا مەدەنىيەت بەلگىسى ۋە جۇڭخۇا مىللىتى ئوبرازىنى ئىپادىلەش ۋە نامايان قىلىشنى ئىلھاملاندۇرۇشى ۋە قوللىشى كېرەك. 

جۇڭخۇا خەلق جۇمھۇرىيىتىنىڭ پۇقرالىرى دۆلەت سىمۋولى ۋە بەلگىسىنىڭ ئىززەت-ھۆرمىتىنى قوغدىشى، جۇڭخۇا مىللىتىنىڭ ئوبرازىنى قوغدىشى، جۇڭخۇا مىللىتىنىڭ شەكىللىنىشى ۋە تەرەققىيات تارىخىنى ھۆرمەتلىشى، ھاقارەت قىلىش، كەمسىتىش ۋە ھۆرمەتسىزلىك قىلىشقا يول قويماسلىقى كېرەك.

**15-ماددا** دۆلەت ئومۇميۈزلۈك ئورتاق قوللىنىلىدىغان دۆلەت تىل -يېزىقىنى ئومۇملاشتۇرىدۇ، مەكتەپ يېشىغا توشمىغان بالىلارنىڭ پۇتۇڭخۇا ئۆگىنىشىنى، مەجبۇرىيەت مائارىپىنى تاماملىغان ئۆسمۈرلەرنىڭ ئاساسەن ئورتاق قوللىنىلىدىغان دۆلەت تىل -يېزىقىنى ئىگىلىشىنى ئىلگىرى سۈرىدۇ. ھەرقانداق تەشكىلات ۋە شەخس پۇقرالارنىڭ ئورتاق قوللىنىلىدىغان دۆلەت تىل-يېزىقىنى ئۆگىنىشى ۋە ئىشلىتىشىگە توسقۇنلۇق قىلىشقا بولمايدۇ. 

دۆلەت ئورگانلىرى ئورتاق قوللىنىلىدىغان دۆلەت تىل -يېزىقىنى رەسمىي تىل -يېزىق قىلىدۇ. ئالاقىدار قانۇن بەلگىلىمىلىرى بويىچە ئاز سانلىق مىللەت تىل -يېزىقىنى ئىشلىتىپ ھۆججەت ئېلان قىلىشقا توغرا كەلگەندە، شۇنىڭ بىلەن بىر ۋاقىتتا ئورتاق قوللىنىلىدىغان دۆلەت تىل -يېزىقى نۇسخىسىنىمۇ تەمىنلىشى كېرەك. 

دۆلەت ئورگانلىرى، ئىجتىمائىي تەشكىلاتلار، كارخانا -كەسپىي ئورۇنلار ۋە باشقا ئىجتىمائىي تەشكىلاتلار ئاممىۋى سورۇنلاردا ئورتاق قوللىنىلىدىغان دۆلەت تىل- يېزىقى بىلەن ئاز سانلىق مىللەت تىل-يېزىقىنى بىر ۋاقىتتا ئىشلىتىشكە توغرا كەلگەندە، ئورنى، تەرتىپى قاتارلىق جەھەتلەردە ئورتاق قوللىنىلىدىغان دۆلەت تىل-يېزىقىنى گەۋدىلەندۈرۈشى كېرەك.

دۆلەت ئاز سانلىق مىللەت تىل -يېزىقىنى ئۆگىنىش ۋە ئىشلىتىشنى ھۆرمەتلەيدۇ ۋە كاپالەتلەندۈرىدۇ، ئاز سانلىق مىللەت تىل -يېزىقىنى قېلىپلاشتۇرۇش، ئۆلچەملەشتۈرۈش ۋە ئۇچۇرلاشتۇرۇش قۇرۇلۇشىنى ئىلگىرى سۈرىدۇ، ئاز سانلىق مىللەتلەرنىڭ قەدىمكى كىتابلىرىنى قوغداش، رەتلەش، تەتقىق قىلىش ۋە ئىشلىتىشنى قوللايدۇ.

**16-ماددا** ھەر دەرىجىلىك ھەر خىل مەكتەپلەر ۋە باشقا مائارىپ ئورگانلىرى دۆلەتنىڭ ئالاقىدار بەلگىلىمىلىرى بويىچە دۆلەت بىر تۇتاش تۈزگەن دەرسلىك ماتېرىياللىرىنى ئىشلىتىشى كېرەك. مائارىپ مەمۇرىي تارماقلىرى دەرسلىك ماتېرىياللىرىنى تۈزۈش، تەكشۈرۈش-بېكىتىشتە جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى ئەمەلىيلەشتۈرۈشى كېرەك. گوۋۇيۈەن مائارىپ مەمۇرىيىتى، مىللىي ئىشلار قاتارلىق تارماقلار جۇڭخۇا مىللىتى ئورتاق گەۋدىسىگە ئائىت بىر يۈرۈش دەرسلىك ماتېرىياللىرىنى تەشكىللەپ تۈزىدۇ.

**17-ماددا** دۆلەت جۇڭگونىڭ مۇستەقىل جۇڭخۇا مىللىتى ئورتاق گەۋدىسى تارىخىي ماتېرىيال سىستېمىسى، سۆز سىستېمىسى، نەزەرىيە سىستېمىسى قۇرۇلۇشىنى ئىلگىرى سۈرىدۇ. ئالىي مەكتەپلەر، ئىلمىي تەتقىقات ئورگانلىرى قاتارلىق ئورۇنلارنىڭ جۇڭخۇا مىللىتى ئورتاق گەۋدىسىگە ئائىت مۇھىم ئاساسىي مەسىلىلەرنى تەتقىق قىلىشنى كۈچەيتىشىنى، جۇڭخۇا مىللىتى ۋە جۇڭخۇا مەدەنىيىتىنىڭ كۆپ خىل بىر گەۋدە شەكلىنىڭ تارىخى ۋە مەزمۇنىنى شەرھلەپ بېرىشىنى، جۇڭخۇا مىللىتىنىڭ شەكىللىنىشى ۋە تەرەققىياتىنىڭ پىرىنسىپى، ئىلمىي ئاساسى، پەلسەپىۋى ئاساسىنى ئاشكارىلىشىنى قوللايدۇ. 

دۆلەت جۇڭخۇا مىللىتى ئورتاق گەۋدىسى نەزەرىيە تەتقىقاتى ئىختىساسلىقلىرىنى يېتىشتۈرۈشنى قوللايدۇ، پەن تۈرلىرىنىڭ تەسىس قىلىنىشى ۋە ئورۇنلاشتۇرۇلۇشىنى ئەلالاشتۇرىدۇ، پەن سىستېمىسى قۇرۇلۇشىنى ئىلگىرى سۈرىدۇ، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى تەتقىقات ئورگانلىرى قۇ رۇلۇشىنى كۈچەيتىدۇ. 

دۆلەت جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئىلمىي ئالاقە سۇپىسى قۇرۇلۇشىنى قوللايدۇ، جۇڭگو ۋە چەت ئەل ئىلمىي ساھەسى، خەلق ئاممىۋى تەشكىلاتلىرى ۋە ئەقىل ئامبىرىنىڭ ئالاقە-ھەمكارلىقنى قانات يايدۇرۇشىنى ئىلگىرى سۈرىدۇ.

**18-ماددا** دۆلەت جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەربىيەسىنى مىللىي مائارىپ، كادىرلار مائارىپى، ئىجتىمائىي مائارىپ سىستېمىسىغا كىرگۈزىدۇ. ھەر دەرىجىلىك ھەر خىل مەكتەپلەر ۋە باشقا مائارىپ ئورگانلىرى جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى پۈتۈن مائارىپ جەريانىغا سىڭدۈرۈشى، دەرسخانا ئوقۇتۇشى، ئىجتىمائىي ئەمەلىيەت، تېمىلىق تەربىيە ۋە تور مائارىپى بىرلەشتۈرۈلگەن مائارىپ سىستېمىسىنى بەرپا قىلىشى كېرەك. 

ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى جۇڭخۇا مۇنەۋۋەر ئەنئەنىۋى مەدەنىيەت بايلىقلىرى، قىزىل بايلىقلار ۋە جۇڭگوچە سوتسىيالىستىك قۇرۇلۇش نەتىجىلىرىنى ئەكس ئەتتۈرىدىغان ھەر خىل بايلىقلاردىن تولۇق پايدىلىنىپ، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەربىيەسىنى كەڭ كۆلەمدە قانات يايدۇرۇشى، شارائىتى بار جايلار يەرلىك شارائىتقا قاراپ جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەربىيە ئەمەلىيەت بازىسىنى بىر تۇتاش قۇرۇشى كېرەك.

**19-ماددا** گېزىت-ژۇرنال، رادىيو، تېلېۋىزىيە، ئىنتېرنېت قاتارلىق تاراتقۇلار جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى قۇرۇلۇشىنى ئىلگىرى سۈرۈشكە ئائىت تەشۋىقات-دوكلات، نەتىجە كۆرسىتىش قاتارلىق خىزمەتلەرنى قانات يايدۇرۇشى كېرەك. 

دۆلەت خەلقئارالىق تارقىتىش ئىقتىدارى قۇرۇلۇشىنى ئىلگىرى سۈرىدۇ، تاشقى ئىنسانپەرۋەرلىك ئالاقىسىنى قانات يايدۇرۇشنى قوللايدۇ، جۇڭخۇا مىللىتى تارىخى ۋە جۇڭخۇا مىللىتى ئورتاق گەۋدىسى نەزەرىيەسىنى شەرھلەپ بېرىدۇ، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى قۇرۇلۇشىنى تونۇشتۇرىدۇ، دۇنيانىڭ جۇڭخۇا مىللىتى ۋە جۇڭخۇا مەدەنىيىتىنى تېخىمۇ ياخشى چۈشىنىشى ۋە بىلىشىنى ئىلگىرى سۈرىدۇ، ئىنسانىيەت مەدەنىيىتىنىڭ ئالاقە-ئۆزئارا ئۆرنەك ئېلىشىنى ئىلگىرى سۈرىدۇ.

**20-ماددا** ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى ئائىلە، ئائىلە تەربىيەسى ۋە ئائىلە ئەنئەنىسى قۇرۇلۇشىغا سىڭدۈرۈشنى ئىلگىرى سۈرۈشى كېرەك. 

قۇرامىغا يەتمىگەنلەرنىڭ ئاتا-ئانىسى ياكى باشقا ۋەسىيلىرى قانۇن بويىچە ئائىلە تەربىيەسى مەسئۇلىيىتىنى ئادا قىلىشى، قۇرامىغا يەتمىگەنلەرنى جۇڭگو كوممۇنىستىك پارتىيەسىنى سۆيۈش، ۋەتەننى سۆيۈش، خەلقنى سۆيۈش، جۇڭخۇا مىللىتىنى سۆيۈشكە تەربىيەلىشى ۋە يېتەكلىشى، جۇڭخۇا مىللىتى بىر ئائىلە كىشىلىرى دېگەن قاراشنى تۇرغۇزۇشى، قۇرامىغا يەتمىگەنلەرگە مىلللەتلەر ئىتتىپاقلىق ى ۋە تەرەققىياتىغا پايدىسىز قاراشلارنى سىڭدۈرمەسلىكى كېرەك.

**21-ماددا** دۆلەت شياڭگاڭ ئالاھىدە مەمۇرىي رايونى، ئاۋمېن ئالاھىدە مەمۇرىي رايونىنىڭ جۇڭخۇا مىللىتى تارىخى، جۇڭخۇا مەدەنىيىتى ۋە دۆلەت ئەھۋالى تەربىيەسىنى قانات يايدۇرۇشىنى قوللايدۇ، شياڭگاڭ ئالاھىدە مەمۇرىي رايونىدىكى قېرىنداشلار، ئاۋمېن ئالاھىدە مەمۇرىي رايونىدىكى قېرىنداشلارنى دۆلەتنىڭ ئىگىلىك ھوقۇقى، بىخەتەرلىكى، تەرەققىيات مەنپەئەتىنى ئاڭلىق قوغداشقا يېتەكلەيدۇ. 

دۆلەت ئىككى قىرغاق ئىقتىساد-مەدەنىيەت ئالاقە-ھەمكارلىقىنى ئىلگىرى سۈرىدۇ، ئىككى قىرغاقتىكى ھەر قايسى ساھەلەرنىڭ يۇغۇرما تەرەققىياتىنى چوڭقۇرلاشتۇرىدۇ، تەيۋەنلىك قېرىنداشلارنىڭ جۇڭخۇا مىللىتىگە بولغان تەۋەلىك تۇيغۇسى، ئېتىراپ قىلىش تۇيغۇسى، شەرەپ تۇيغۇسىنى ئاشۇرىدۇ، ئىككى قىرغاقتىكى قېرىنداشلارنىڭ ئورتاق جۇڭخۇا مەدەنىيىتىنى ئەۋج ئالدۇرۇشىنى، جۇڭخۇا مىللىتىگە تەۋە، جۇڭگولۇق دېگەن تونۇشنى كۈچەيتىشىنى ئىلگىرى سۈرىدۇ. 

دۆلەت چەت ئەلدىكى جۇڭگولۇقلار بىلەن بولغان ئالاقىنى كۈچەيتىدۇ، چەت ئەلدىكى جۇڭگولۇقلارنىڭ جۇڭگو-چەت ئەل مەدەنىيەت ئالاقە-ھەمكارلىقىنى ئىلگىرى سۈرۈشىنى قوللايدۇ.

## 3-باب ئالاقە، ئالماشتۇرۇش، يۇغۇرۇلۇشنى ئىلگىرى سۈرۈش

**22-ماددا** ناھىيە دەرىجىلىكتىن يۇقىرى خەلق ھۆكۈمەتلىرى ئىقتىساد-ئىجتىمائىي تەرەققىيات پىلانى ۋە ئاممىۋى بايلىقلارنى تەقسىملەشنى بىر تۇتاش پىلانلاپ، ئۆزئارا گىرەلىشىپ كەتكەن مەھەللە مۇھىتى قۇرۇلۇشىنى ئىلگىرى سۈرىدۇ، ھەر مىللەت ئاممىسىنىڭ ئورتاق ئولتۇراقلىشىشى، ئورتاق ئۆگىنىشى، ئورتاق قۇرۇشى، ئورتاق بەھرىلىنىشى، ئورتاق ئىش قىلىشى، ئورتاق خۇشاللىنىشىغا پايدىلىق ئىجتىمائىي شارائىتنى مۇكەممەللەشتۈرىدۇ.

**23-ماددا** ناھىيە دەرىجىلىكتىن يۇقىرى يەرلىك خەلق ھۆكۈمەتلىرى جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى شەھەر پىلانى، قۇرۇلۇشى، باشقۇرۇشى ۋە مۇلازىمىتىنىڭ پۈتۈن جەريانىغا كىرگۈزۈپ، يەرلىك شارائىتقا ماس ھالدا دوستلۇق تىپىدىكى، سىغدۇرۇشچان، بىرلىشىشچان شەھەر مىللەت خىزمىتى سىياسەت تەدبىرلىرىنى مۇكەممەللەشتۈرۈشى كېرەك. 

ھەر دەرىجىلىك يەرلىك خەلق ھۆكۈمەتلىرى يەرلىك ئەمەلىي ئەھۋالغا ئاساسەن، شەھەر -يېزا قۇرۇلۇشى، نوپۇس باشقۇرۇش، تۇرالغۇ سىياسىتى، ئىشقا ئورۇنلىشىش ۋە ئىگىلىك تىكلەش، ئىجتىمائىي مۇلازىمەت قاتارلىق جەھەتلەردە كونكرېت تەدبىرلەرنى قوللىنىپ، ھەر مىللەتنىڭ ئىتتىپاقلىشىپ ئۆزئارا يۇغۇرۇلۇشىنى ئىلگىرى سۈرۈشى كېرەك. 

ھەر دەرىجىلىك يەرلىك خەلق ھۆكۈمەتلىرى ھەر مىللەت ئاممىسىنى مەھەللە قۇرۇلۇشى، مەھەللە ئىدارە قىلىنىشى، مەھەللە پائالىيەتلىرىگە ئورتاق قاتنىشىشقا تەشكىللىشى ۋە يېتەكلىشى، ئالاقىدار ئورۇن ۋە تەشكىلاتلارنىڭ ئىجتىمائىي مۇلازىمەت بىلەن تەمىنلىشىنى قوللاپ، ھەر مىللەت ئاممىسىنىڭ ئىناق ئۆزئارا يۇغۇرۇلۇشىنى ئىلگىرى سۈرۈشى كېرەك.

**24-ماددا** ناھىيە دەرىجىلىكتىن يۇقىرى خەلق ھۆكۈمەتلىرى مىللىي رايونلار ئارا، مىللىي رايونلار بىلەن باشقا رايونلار ئارا نوپۇس ئېقىمى مۇلازىمەت سۇپىسىنى ئورتاق قۇرۇش ۋە ئۇچۇرلارنى ئورتاق بەھرىلىنىشنى كۈچەيتىپ، ھەمكارلىق ئىقتىدارىنى ئاشۇرۇپ، رايون ھالقىغان ھۆكۈمەت ئىشل ىرى مۇلازىمىتىنىڭ قولايلىق دەرىجىسىنى ئۆستۈرۈشى كېرەك.

**25-ماددا** ناھىيە دەرىجىلىكتىن يۇقىرى خەلق ھۆكۈمەتلىرى قانۇن بويىچە مىللىي رايونلار ئارا، مىللىي رايونلار بىلەن باشقا رايونلار ئارا رايون ھالقىغان ئىشقا ئورۇنلاشقان ۋە ئىگىلىك تىكلىگەن پۇقرالارنىڭ قانۇنلۇق ھوقۇق-مەنپەئىتىگە كاپالەتلىك قىلىشى، قانۇن-نىزام ۋە سىياسەت، دۆلەت ئورتاق تىل-يېزىقى، كەسپىي ماھارەت قاتارلىق جەھەتلەردىكى تەربىيەلەشنى قانات يايدۇرۇشنى قوللاپ، كەسپىي يېتەكلەش، كەسپىي تونۇشتۇرۇش قاتارلىق مۇلازىمەتلەرنى قانات يايدۇرۇشى كېرەك.

**26-ماددا** گوۋۇيۈەن مائارىپ مەمۇرىي تارمىقى ۋە ئۆلكە دەرىجىلىك خەلق ھۆكۈمەتلىرى مىللىي رايونلار ۋە باشقا رايونلاردىكى ئالىي مەكتەپلەرنىڭ ئىككى يۆنىلىشلىك رايون ھالقىغان ئوقۇغۇچى قوبۇل قىلىشىنى قوللاپ، ئىختىساسلىقلارنى يېتىشتۈرۈشتىكى ھەمكارلىقنى كۈچەيتىشى كېرەك. 

گوۋۇيۈەن مائارىپ مەمۇرىي تارمىقى ۋە ناھىيە دەرىجىلىكتىن يۇقىرى يەرلىك خەلق ھۆكۈمەتلىرى مىللىي رايونلار ۋە باشقا رايونلاردىكى ئوقۇتقۇچىلارنىڭ ئالماشتۇرۇشىنى ئىلھاملاندۇرۇشى ۋە قوللىشى كېرەك. 

ھەر دەرىجىلىك ۋە تۈرلۈك مەكتەپلەر ھەم باشقا مائارىپ ئاپپاراتلىرى مەكتەپ ۋە ئوقۇغۇچىلارنىڭ ئالاھىدىلىكىنى بىرلەشتۈرۈپ، ھەر مىللەت ئوقۇغۇچىلىرىنىڭ بىرلىكتە ئۆگىنىشى، بىرلىكتە تۇرمۇش كەچۈرۈشى، بىرلىكتە ئۆسۈپ يېتىلىشىنى ئىلگىرى سۈرۈشى كېرەك.

**27-ماددا** ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى مەكتەپلەر، ئاممىۋى تەشكىلاتلار ۋە باشقا ئىجتىمائىي تەشكىلاتلارنىڭ جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسىنىڭ شەكىللىنىشى ۋە تەرەققىياتىدىكى تارىخىي مەدەنىيەت بايلىقلىرىدىن پايدىلىنىپ، ئۆسمۈرلەرنىڭ رايون ھالقىغان ئىجتىمائىي ئەمەلىيەت، ئۆگىنىش ساياھىتى ۋە زىيارەت قىلىپ تەكشۈرۈش قاتارلىق ئالاقە پائالىيەتلىرىنى قانات يايدۇرۇشىنى قوللاپ، مىللىي ئىپتىخارلىق تۇيغۇسى ۋە ئىشەنچىنى ئاشۇرۇشى كېرەك.

**28-ماددا** ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى پىدائىيلار ۋە پىدائىيلار مۇلازىمەت تەشكىلاتلىرىنىڭ تۈرلۈك پىدائىيلار مۇلازىمەت پائالىيەتلىرىدە ھەر مىللەت ئاممىسىنىڭ ئالاقە ھەمكارلىقى ۋە دوستلۇق ئۆزئارا ياردىمىنى ئىلگىرى سۈرۈشىنى ئىلھاملاندۇرۇشى ۋە قوللىشى كېرەك.

**29-ماددا** دۆلەت ھەر مىللەت مەدەنىيىتىنىڭ ئۆزئارا ئۆرنەك ئېلىپ ئۆزئارا يۇغۇرۇلۇشىنى ئىلگىرى سۈرۈپ، ھەر مىللەتنىڭ مۇنەۋۋەر ئەنئەنىۋى مەدەنىيەتلىرىدىن ئۆزئارا ھۇزۇرلىنىشىنى، تىل -يېزىقنى ئۆزئارا ئۆگىنىشىنى ئىلھاملاندۇرىدۇ. ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى مەدەنىيەت خىزمەتچىلىرى ۋە ئالاقىدار ئورۇنلارنىڭ جۇڭخۇا مەدەنىيىتىنىڭ چوڭقۇر يىلتىزىنى ئەكس ئەتتۈرىدىغان، ھەر مىللەتنىڭ ئۆزئارا ئالاقە، ئۆزئارا يۇغۇرۇلۇشىنى گەۋدىلەندۈرىدىغان ئەدەبىيات-سەنئەت ئەسەرلىرىنى ئىجاد قىلىشى ۋە نامايان قىلىشىنى قوللىشى كېرەك. 

ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى ك ۇتۇ پخانا، مۇزېي، مەدەنىيەت يادىكارلىقلىرىنى ساقلاش ئورنى، مەدەنىيەت سارىيى (پونكىتى)، خاتىرە سارىيى، گۈزەل-سەنئەت سارىيى، پەن-تېخنىكا سارىيى، ئىشچىلار مەدەنىيەت سارىيى، ئۆسمۈرلەر سارىيى قاتارلىق ئاممىۋى مەدەنىيەت مۇلازىمەت ئورۇنلىرىنىڭ جۇڭخۇا مىللەتلىرىنىڭ تا رىخى ۋە دۆلەتنىڭ گۈللىنىپ تەرەققىي قىلىشى قاتارلىق جەھەتلەردىكى مەزمۇنلارنى ئەكس ئەتتۈرىدىغان نامايان قىلىش ۋە ئالاقە پائالىيەتلىرىنى قانات يايدۇرۇشىنى قوللىشى كېرەك. 

ھەر دەرىجىلىك خەلق ھۆكۈمەتلىرى جۇڭخۇا مىللەتلىرىنىڭ مول مەدەنىيەت، تەنتەربىيە قاتارلىق تۈرلۈك بايلىقلىرىدىن پايدىلىنىپ، ھەر مىللەت ئاممىسى ياقتۇرىدىغان، ئورتاق قاتنىشىدىغان جۇڭخۇا مىللەتلىرىنىڭ ئەنئەنىۋى بايراملىرى، خەلق ئۆرپ-ئادەت مەدەنىيىتى، تەنتەربىيە مۇسابىقىلىرى قاتارلىق ئالاقە پائالىيەتلىرىنى ئۆتكۈزۈشنى ئىلھاملاندۇرۇشى ۋە قوللىشى كېرەك.

**30-ماددا** دۆلەت ساياھەتنىڭ ھەر مىللەتنىڭ ئۆزئارا ئالاقە، ئۆزئارا يۇغۇرۇلۇشىنى ئىلگىرى سۈرۈشتىكى ئاكتىپ رولىنى جارى قىلدۇرۇپ، مەدەنىيەت بىلەن ساياھەتنىڭ چوڭقۇر يۇغۇرۇلۇپ تەرەققىي قىلىشىنى ئىلگىرى سۈرۈپ، جۇڭخۇا مەدەنىيىتىنى ئەۋج ئالدۇرۇشقا پايدىلىق ساياھەت لىنىيەسى ۋ ە مەدەنىيەت ئىجادىيەت مەھسۇلاتلىرىنى ئېچىشى كېرەك. 

ناھىيە دەرىجىلىكتىن يۇقىرى خەلق ھۆكۈمەتلىرىنىڭ مىللەت خىزمىتى تارماقلىرى مەدەنىيەت ۋە ساياھەت، مەدەنىيەت يادىكارلىقلىرى قاتارلىق تارماقلار بىلەن بىرلىكتە، جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى نەزەرىيەسى ۋە تارىخىي ماتېرىياللىرىدىن پايدىلىنىپ، كۆرگەزمە، چۈشەندۈ رۈش قاتارلىق مەزمۇنلارنى قېلىپلاشتۇرۇشى كېرەك

**31-ماددا** دۆلەت ئىنتېرنېت، چوڭ سانلىق مەلۇمات، سۈنئىي ئەقىل قاتارلىق زامانىۋى تېخنىكا ۋاسىتىلىرىدىن پايدىلىنىپ ئالاقە پائالىيەتلىرىنى قانات يايدۇرۇشنى قوللاپ، تور مەھسۇلاتلىرى، مۇلازىمەت بىلەن تەمىنلىگۈچىلەرنى جۇڭخۇا مىللەتلىرىنىڭ بۈيۈك ئىتتىپاقلىقىنى گەۋدىلەندۈرىدىغان ئەسەر ۋە ئۇچۇرلارنى ئىشلەپ تارقىتىشقا ئىلھاملاندۇرىدۇ ۋە يېتەكلەيدۇ، جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەشكە پايدىلىق ئىناق تور مۇھىتىنى يارىتىدۇ. ھەرقانداق تەشكىلات ۋە شەخسنىڭ يېزىق، رەسىم ۋە ئاۋازلىق-كۆرۈنۈشلۈك ماتېرىيال قاتارلىق ئۇسۇللار بىلەن مىللىي ئۆچمەنلىك، مىللىي كەمسىتىش قاتارلىق مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتىغا بۇزغۇنچىلىق قىلىدىغان مەزمۇندىكى ئۇچۇرلارنى ئىشلەپ تارقىتىشىغا يول قويۇلمايدۇ. 

تور تىجارەتچىلىرى ئۆز ئىشلەتكۈچىلىرى تارقاتقان ئۇچۇرلارنى باشقۇرۇشنى كۈچەيتىشى، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىيات ىغا بۇزغۇنچىلىق قىلىدىغان مەزمۇندىكى ئۇچۇرلارنى بايقىغان ھامان قانۇن بويىچە دەرھال بىر تەرەپ قىلىش تەدبىرى قوللىنىپ، ئۇچۇرلارنىڭ تارقىلىشىنىڭ ئالدىنى ئېلىشى، ئالاقىدار خاتىرىلەرنى ساقلىشى ھەمدە ئالاقىدار مەسئۇل تارماقلارغا دوكلات قىلىشى كېرەك.

## 4-باب ئورتاق گۈللىنىش ۋە تەرەققىياتنى ئىلگىرى سۈرۈش

**32-ماددا** دۆلەت يېڭى تەرەققىيات ئىدىيەسىنى ئىزچىللاشتۇرۇپ، مىللىي رايونلارنىڭ ئىسلاھات، ئېچىۋېتىشنى ئومۇميۈزلۈك چوڭقۇرلاشتۇرۇشىنى، دۆلەت تەرەققىيات ئىستراتېگىيەسىگە ئومۇميۈزلۈك سىڭىشىنى قوللاپ، ئۆزىنى تەرەققىي قىلدۇرۇش ئىقتىدارىنى ئۆستۈرۈپ، مىللىي رايونلارنىڭ يۇقىر ى سۈپەتلىك تەرەققىياتىنى تېزلىتىپ، ھەر مىللەتنىڭ ئورتاق بېيىشىنى ئىلگىرى سۈرۈپ، ھەر مىللەتنىڭ سوتسىيالىستىك زامانىۋىلاشتۇرۇشقا ئورتاق يۈزلىنىشىنى ئىلگىرى سۈرىدۇ.

**33-ماددا** ناھىيە دەرىجىلىكتىن يۇقىرى خەلق ھۆكۈمەتلىرى ۋە ئۇنىڭ ئالاقىدار تارماقلىرى ئىقتىساد، جەمئىيەت تەرەققىيات ساھەسىدىكى سىياسەت پىلانىنى تۈزۈش ۋە يولغا قويۇشتا، جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەشكە پايدىلىق بولۇشى، دۆلەتنىڭ بىرلىكىنى قوغداش، بۆلگۈنچىلىككە قارشى تۇرۇشقا پايدىلىق بولۇشى، خەلق تۇرمۇشىنى ياخشىلاش، كىشىلەر قەلبىنى مۇجەسسەملەشكە پايدىلىق بولۇشى كېرەك.

**34-ماددا** دۆلەت رايونلارنىڭ ماس تەرەققىياتىنى ئىلگىرى سۈرۈپ، رايونلارنىڭ بىر گەۋدىلەشكەن تەرەققىيات مېخانىزمى ۋە پەرقلىق رايونلارنى قوللاش سىياسىتىنى مۇكەممەللەشتۈرۈپ، ياردەم بېرىش - ھەمكارلىق مېخانىزمىنى ساغلاملاشتۇرىدۇ. 

دۆلەت مىللىي رايونلارنىڭ رايون ئاساسىي فۇنكسىيە ئورنىنى بىرلەشتۈرۈپ، دۆلەت چېگرا مۇداپىئە بىخەتەرلىكى، بايلىق- ئېنېرگىيە بىخەتەرلىكى، ئاشلىق بىخەتەرلىكى ۋە ئېكولوگىيە بىخەتەرلىكى قاتارلىق جەھەتلەردە بۇرچ ۋە مەسئۇلىيەتنى زىممىسىگە ئېلىپ، رولىنى تولۇق جارى قىلدۇرۇشىنى قوللايدۇ. 

دۆلەت مىللىي رايونلارنىڭ ئۆزىنىڭ ئەۋزەللىكىنى جارى قىلدۇرۇپ، دۆلەت ئىچى چوڭ ئايلىنىشنى ئاساس قىلغان، دۆلەت ئىچى ۋە خەلقئارا قوش ئايلىنىش ئۆزئارا ئىلگىرى سۈرىدىغان يېڭى تەرەققىيات ئەندىزىسىنى بەرپا قىلىشقا مۇلازىمەت قىلىشىنى،سۈپەتلىك ئورتاق قۇرۇشقا چوڭقۇر سىڭىشىنى قوللايدۇ.

**35-ماددا** گوۋۇيۈەننىڭ ئالاقىدار تارماقلىرى ۋە ناھىيە دەرىجىلىكتىن يۇقىرى يەرلىك خەلق ھۆكۈمەتلىرى قاتناش، ئېنېرگىيە، سۇچىلىق، ئۇچۇر، ئەشيا ئوبوروتى قاتارلىق ئۇل ئەسلىھە قۇرۇلۇشىنى قوللاپ، مىللىي رايونلار بىلەن باشقا رايونلار ئارا ئۆزئارا تۇتاشتۇرۇشنى كۈچەيتىشى كېرەك.

**36-ماددا** دۆلەت زامانىۋىلاشقان كەسىپ سىستېمىسىنى مۇكەممەللەشتۈرۈپ، يەرلىك شارائىتقا ماس ھالدا يېڭى سۈپەتلىك ئىشلەپچىقىرىش كۈچىنى تەرەققىي قىلدۇرۇپ، رايونلار ئارا كەسىپ ھەمكارلىقى ۋە مەنپەئەتنى ئورتاق بەھرىلىنىش مېخانىزمى قۇرۇلۇشىنى تەرتىپلىك ئىلگىرى سۈرۈپ، مىللىي رايونلارنىڭ رايون كەسىپ زەنجىرى ۋە تەمىنلەش زەنجىرى ئورۇنلاشتۇرمىسىنى ئەلالاشتۇرۇش، مەملىكەت بويىچە بىرلىككە كەلگەن چوڭ بازار قۇرۇلۇشىنى ئىلگىرى سۈرۈشتە رولىنى تولۇق جارى قىلدۇرۇشىنى قوللايدۇ. 

گوۋۇيۈەننىڭ ئالاقىدار تارماقلىرى ۋە ناھىيە دەرىجىلىكتىن يۇقىرى يەرلىك خەلق ھۆكۈمەتلىرى مىللىي رايونلارنىڭ بايلىق ئەۋزەللىكىگە تايىنىپ ھەمدە زامانىۋى پەن -تېخنىكىدىن پايدىلىنىپ، دېھقانچىلىق، ئورمانچىلىق، چارۋىچىلىق، بېلىقچىلىق، يېزا ئىگىلىك قوشۇمچە مەھسۇلاتل ىرىنى پىششىقلاپ ئىشلەش كەسپى، مەدەنىيەت-سەنئەت كەسپى، شۇنداقلا ئەنئەنىۋى قول ھۈنەرۋەنچىلىك، ئەنئەنىۋى تېبابەت قاتارلىق ئالاھىدە ئەۋزەل كەسىپلەرنى تەرەققىي قىلدۇرۇشىنى، يېڭى تىپتىكى يېزا كوللېكتىپ ئىگىلىكىنى تەرەققىي قىلدۇرۇشىنى، يېزا-كەنتلەرنى ئومۇميۈزلۈك گۈللەندۈرۈش ۋە شەھەر- يېزا بىرلىشىپ تەرەققىي قىلىشنى ئىلگىرى سۈرۈشىنى قوللايدۇ.

**37-ماددا** گوۋۇيۈەننىڭ ئالاقىدار تارماقلىرى ۋە ناھىيە دەرىجىلىكتىن يۇقىرى يەرلىك خەلق ھۆكۈمەتلىرى ئاممىۋى مۇلازىمەت بايلىقلىرىنى مۇۋاپىق تەقسىملەشنى ئىلگىرى سۈرۈپ، ئاساسىي ئاممىۋى مۇلازىمەتنىڭ تەڭپۇڭلۇقى ۋە قولغا كەلتۈرۈشچانلىقىنى ئاشۇرۇپ، مىللىي رايونلاردىكى ھەر دەرىجىلىك ۋە تۈرلۈك مائارىپنىڭ ماس تەرەققىياتىنى ئىلگىرى سۈرۈپ، مىللىي رايونلارنىڭ ئاممىۋى مۇلازىمەت كاپالەت ئىقتىدارىنى ئۆستۈرۈشى كېرەك.

**38-ماددا** گوۋۇيۈەننىڭ ئالاقىدار تارماقلىرى ۋە ناھىيە دەرىجىلىكتىن يۇقىرى يەرلىك خەلق ھۆكۈمەتلىرى ئېكولوگىيە، كەسىپ، شەھەر قاتارلىق تۈرلۈك بوشلۇق ئورۇنلاشتۇرمىسىنى بىر تۇتاش پىلانلاپ، مىللىي رايونلارنىڭ ئېكولوگىيەلىك مۇھىتىنى قوغداش ۋە تەبىئىي بايلىقلىرىنى مۇۋاپىق ئېچىش ۋە ئىشلىتىشنى كۈچەيتىپ، ھەر مىللەت ئاممىسىنى جۇڭخۇا مىللەتلىرىنىڭ مەڭگۈلۈك تەرەققىياتىدىكى ئېكولوگىيەلىك بوشلۇقنى ئورتاق قوغداشقا يېتەكلىشى كېرەك

**39-ماددا** گوۋۇيۈەننىڭ ئالاقىدار تارماقلىرى ۋە چېگرا رايونلاردىكى ناھىيە دەرىجىلىكتىن يۇقىرى يەرلىك خەلق ھۆكۈمەتلىرى چېگرانى گۈللەندۈرۈپ خەلقنى بېيىتىش، چېگرانى مۇقىملاشتۇرۇپ چېگرانى مۇستەھكەملەشنى بىر تۇتاش ئىلگىرى سۈرۈپ، چېگرا رايونلارنىڭ قۇرۇلۇشىنى قوللاشنى ئىلھا ملاندۇرۇپ، ئۇل ئەسلىھە ۋە ئاممىۋى مۇلازىمەت ئەسلىھەلىرى قۇرۇلۇشىنى ئىلگىرى سۈرۈپ، چېگرا يېزىلىرىنىڭ ئولتۇراقلىشىش مۇھىتى ۋە ئىشلەپچىقىرىش-تۇرمۇش شارائىتىنى ياخشىلىشى كېرەك. 

گوۋۇيۈەننىڭ ئالاقىدار تارماقلىرى ۋە چېگرا رايونلاردىكى ناھىيە دەرىجىلىكتىن يۇقىرى يەرلىك خەلق ھۆكۈمەتلىرى چېگرا بويىدىكى شەھەر-بازار سىستېمىسى قۇرۇلۇشىنى كۈچەيتىپ، ئېچىش-ئېچىۋېتىش سىياسىتىنى ساغلاملاشتۇرۇپ، ئېچىش -ئېچىۋېتىش سۇپىسى ۋە كەسىپ سۇپىسى ئەسلىھەلىر ى قۇرۇلۇشىنى ئىلگىرى سۈرۈپ، كەسىپ تىرەك ئىقتىدارىنى ئاشۇرۇپ، چېگرا سودىسىنى قوللاپ، چېگرا ساياھىتىنى تەرەققىي قىلدۇرۇپ، چېگرا ھالقىغان ئىقتىسادىي ھەمكارلىقنى ئىلھاملاندۇرۇشى كېرەك.

**40-ماددا** دۆلەت دەۋرنىڭ يېڭىچە روھىنى يېتىلدۈرۈشنى كۈچەيتىپ، پۇقرالار ئەخلاق قۇرۇلۇشىنى ئىلگىرى سۈرۈپ، قانۇنچىلىق تەشۋىق-تەربىيەسى ۋە ئىلىم -پەن ئومۇملاشتۇرۇشنى قانات يايدۇرۇپ، ئۆرپ-ئادەتنى ئۆزگەرتىپ، مەدەنىيەتلىك ئىلغار يېڭى ئادەتنى تەشەببۇس قىلىپ، ھەر مىللەت ئاممىسىنى توي - تۆكۈن، نەزىر -چىراغ قاتارلىق پائالىيەتلەردە قانۇنغا ئاڭلىق رىئايە قىلىشقا، ئاممىۋى تەرتىپ ۋە ياخشى ئۆرپ-ئادەتكە بويسۇنۇشقا يېتەكلەيدۇ. دۆلەت پۇقرالارنىڭ نىكاھ ئەركىنلىكىنى قوغدايدۇ. ھەرقانداق تەشكىلات ۋە شەخسنىڭ مىللىي سالاھىيەت، ئۆرپ-ئادەت، دىنىي ئېتىقاد قاتارلىقلارنى باھانە قىلىپ نىكاھ ئەركىنلىكىگە ئارىلىشىشىغا يول قويۇلمايدۇ.

## 5-باب كاپالەت ۋە نازارەت قىلىش

**41-ماددا** پارتىيە رەھبەرلىكىنى بىر تۇتاش يۈرگۈزۈش، ھۆكۈمەتنىڭ قانۇن بويىچە باشقۇرۇشى، بىرلىك سەپ خىزمىتى تارماقلىرىنىڭ باشلامچىلىق قىلىپ ماسلاشتۇرۇشى، مىللەت خىزمىتى تارماقلىرىنىڭ بۇرچىنى ئادا قىلىپ مەسئۇلىيىتىنى ئۈستىگە ئېلىشى، ھەر قايسى تارماقلارنىڭ ماسلىشىپ ھەمك ارلىشىشى، پۈتۈن جەمئىيەتنىڭ ئورتاق قاتنىشىشىدىن ئىبارەت مىللەت خىزمىتى ئەندىزىسىنى بوشاشماي ساقلاش ۋە مۇكەممەللەشتۈرۈش، مىللەت خىزمىتىنى ماسلاشتۇرۇش مېخانىزمىنى ساغلاملاشتۇرۇش كېرەك. 

مەركىزىي بىرلىك سەپ خىزمىتى تارماقلىرى، گوۋۇيۈەننىڭ مىللەتلەر خىزمىتى تارماقلىرى مەملىكەت بويىچە جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، مىللەتلەر ئىتتىپاقلىق ى ۋە تەرەققىياتنى ئىلگىرى سۈرۈش قاتارلىق خىزمەتلەرنى بىر تۇتاش ماسلاشتۇرۇش ۋە ئىزچىللاشتۇرۇشقا مەسئۇل بولىدۇ. ھەر دەرىجىلىك يەرلىك بىرلىك سەپ خىزمىتى تارماقلىرى، مىللەت خىزمىتى تارماقلىرى ئۆز رايونىدىكى جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتنى ئىلگىرى سۈرۈش قاتارلىق خىزمەتلەرنى ماسلاشتۇرۇپ ئىلگىرى سۈرۈش، تەشكىللەپ يولغا قويۇش ۋە ئىزچىللاشتۇرۇشقا مەسئۇل بولىدۇ.

**42-ماددا** مەركىزىي ۋە دۆلەت ئورگانلىرىنىڭ ھەر قايسى تارماقلىرى، يەرلىك ئالاقىدار ئورگانلار ئۆز مەسئۇلىيەت دائىرىسىدە جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتنى ئىلگىرى سۈرۈش خىزمىتىنى قانات يايدۇرىدۇ. ھەر قايسى ئورگانلار ئۆز ئورنىدا يۈز بەرگەن مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتقا بۇزغۇنچىلىق قىلىدىغان قىلمىشلارنى ۋاقتىدا توختىتىشى كېرەك. 

جامائەت خادىملىرى بۇرچىنى ئادا قىلىش ۋە كۈندىلىك تۇرمۇشتا، جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتنى ئىلگىرى سۈرۈشتە ئۈلگىلىك باشلامچى رول ئوينىشى كېرەك.

**43-ماددا** ھەر دەرىجىلىك خەلق ۋەكىللەر قۇرۇلتايلىرى ۋە ناھىيە دەرىجىلىكتىن يۇقىرى خەلق ۋەكىللەر قۇرۇلتايلىرىنىڭ دائىمىي كومىتېتلىرى قانۇندا بەلگىلەنگەن خىزمەت ھوقۇقىغا ئاساسەن، ھەر تۈرلۈك خىزمەتلەردە جۇڭخۇا مىللىتى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، جۇڭخۇا مىللىتى ئورتاق گەۋدىسى قۇرۇلۇشىنى ئىلگىرى سۈرۈش تەلىپىنى ئەمەلىيلەشتۈرىدۇ.

**44-ماددا** ئىشچىلار ئۇيۇشمىسى، كوممۇنىستىك ياشلار ئىتتىپاقى، ئاياللار بىرلەشمىسى، سانائەت-سودا بىرلەشمىسى، ئەدەبىيات -سەنئەتچىلەر بىرلەشمىسى، يازغۇچىلار جەمئىيىتى، پەن - تېخنىكا جەمئىيىتى، چەت ئەلدىن قايتىپ كەلگەن جۇڭگولۇقلار بىرلەشمىسى، تەيۋەنلىك قېرىنداشلار دوستلۇق ج ەمئىيىتى، مېيىپلار بىرلەشمىسى ۋە باشقا ئاممىۋى تەشكىلاتلار ئۆزىنىڭ ئەۋزەللىكىنى جارى قىلدۇرۇپ، ئۆزى ئالاقىلىشىدىغان ساھەلەردە جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش خىزمىتىنى قانات يايدۇرۇپ، ئۆزى ئالاقىلىشىدىغان كوللېكتىپلارنى جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى قۇرۇلۇشىغا كۈچ قوشۇشقا ئىتتىپاقلاشتۇرۇشى كېرەك.

**45-ماددا** كارخانا-كەسپىي ئورۇنلار ئىجتىمائىي ئەخلاققا رىئايە قىلىشى، ئىجتىمائىي مەسئۇلىيەتنى ئادا قىلىشى، جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى كەسپىي تەربىيەلەش، مەدەنىيەت قۇرۇلۇشى قاتارلىق پائالىيەتلەرگە سىڭدۈرۈپ، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتنى ئىلگىرى سۈرۈشى كېرەك. 

كەسىپ جەمئىيەتلىرى، سودا ئۇيۇشمىلىرى، ئىلىم جەمئىيەتلىرى ۋە فوندى جەمئىيەتلىرى قاتارلىق ئىجتىمائىي تەشكىلاتلار جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى كەسپىي تەلەپچانلىق قائىدىسى، كەسپىي ئەخلاق ئۆلچىمى ياكى تەشكىلات نىزامنامىسىدە گەۋدىلەندۈرۈپ، كەسپىي خىزمەتنى بىرلەشتۈرۈپ مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتنى ئىلگىرى سۈرۈشى كېرەك.

**46-ماددا** دىنىي تەشكىلاتلار، دىنىي مەكتەپلەر ۋە دىنىي پائالىيەت سورۇنلىرى دۆلىتىمىزنىڭ دىننى جۇڭگوچىلاشتۇرۇش يۆنىلىشىدە چىڭ تۇرۇپ، جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەشۋىق -تەربىيەسىنى قانات يايدۇرۇپ، دىنغا ئېتىقاد قىلىدىغان ئاممىنى ۋەتەنپەرۋەرلىك ئەنئەنىسىنى ئەۋج ئالدۇرۇشقا يېتەكلەپ، مىللەتلەرنىڭ ئىناقلىقىنى، دىنلارنىڭ ئىناقلىقىنى، جەمئىيەتنىڭ ئىناقلىقىنى ئىلگىرى سۈرۈشى كېرەك.

**47-ماددا** ئاساسىي قاتلام خەلق ھۆكۈمەتلىرى ئاھالىلەر كومىتېتى، كەنت ئاھالىلەر كومىتېتىنىڭ جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش خىزمىتىگە يېتەكچىلىك قىلىشى، قوللىشى ۋە ياردەم بېرىشى كېرەك. 

ئاھالىلەر كومىتېتى، كەنت ئاھالىلەر كومىتېتى ئاھالىلەر ئەھدىنامىسى، كەنت نىزامنامىسىدە جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى گەۋدىلەندۈرۈپ، ھەر مىللەت ئاممىسىنى ئىتتىپاقلىقنى ئاشۇرۇشقا، بىرلىكتە ئىلگىرىلەشكە يېتەكلىشى كېرەك.

**48-ماددا** ئارمىيە بۇ قانۇن ۋە ئالاقىدار قانۇن بەلگىلىمىلىرى بويىچە، دۆلەت مۇداپىئە پائالىيەتلىرىنى بىرلەشتۈرۈپ جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەشۋىق- تەربىيەسىنى قانات يايدۇرۇپ، ئۆزىنىڭ بايلىقلىرىدىن تولۇق پايدىلىنىپ جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى قۇرۇلۇشىغا كاپالەتلىك قىلىپ، ھەربىي - ھۆكۈمەت، ھەربىي -خەلق ئىتتىپاقلىقىنى مۇستەھكەملەپ ۋە راۋاجلاندۇرۇپ، دۆلەتنىڭ بىرلىكى ۋە بىخەتەرلىكىنى قوغدىشى كېرەك.

**49-ماددا** دۆلەت جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى كادىرلارنى تەربىيەلەش، تاللاش، ئىشلىتىش ۋە باشقۇرۇشنىڭ پۈتۈن جەريانىغا سىڭدۈرۈپ، ئاز سانلىق مىللەت كادىرلىرىنى تەربىيەلەش ۋە ئىشلىتىشكە ئەھمىيەت بېرىدۇ. 

دۆلەت مىللىي رايونلاردىكى كەسپىي تېخنىكا ئىختىساسلىقلىرى ۋە يۇقىرى ماھارەتلىك ئىختىساسلىقلارنى يېتىشتۈرۈشنى قوللاپ، مىللىي رايونلار ئارا، مىللىي رايونلار بىلەن باشقا رايونلار ئارا كادىرلار ۋە ئىختىساسلىقلار ئالماشتۇرۇشنى ئىلگىرى سۈرۈپ، كادىرلار ۋە ئىختىساس لىقلارنى مىللىي رايونلارنىڭ ئاساسىي قاتلىمىغا يۆتكىلىشكە يېتەكلەيدۇ.

**50-ماددا** ناھىيە دەرىجىلىكتىن يۇقىرى خەلق ھۆكۈمەتلىرى جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىيات خىزمىتىنى خەلق ئىگىلىكى ۋە ئىجتىمائىي تەرەققىيات پىلانى ھەم يىللىق پىلانغا كىرگۈزۈپ، ئالاقىدار خىزمەت خىراجىتىنى خامچوتقا كىرگۈزۈشى كېرەك. 

ناھىيە دەرىجىلىكتىن يۇقىرى خەلق ھۆكۈمەتلىرىنىڭ مالىيە تارماقلىرى قانۇن بويىچە مەبلەغ ئىشلىتىش قاتارلىق ئەھۋاللارنى يېتەكلەش ۋە نازارەت قىلىشنى قانات يايدۇرىدۇ

**51-ماددا** دۆلەت مىللىي ئىشلارنى ئورتاق قۇرۇش، ئورتاق ئىدارە قىلىش، ئورتاق بەھرىلىنىش ئىجتىمائىي ئىدارە قىلىش مېخانىزمىغا كىرگۈزۈپ، تەشكىللەپ ماسلاشتۇرۇش ۋە ئاساسىي قاتلامنى ئىدارە قىلىش كۈچىنى كۈچەيتىپ، پەن-تېخنىكا تىرەكلىكىنى كۈچەيتىپ، ئىجتىمائىي ئىدارە قىلىش ئۈنۈمىنى ئۆستۈرىدۇ.

**52-ماددا** گوۋۇيۈەننىڭ ئالاقىدار تارماقلىرى ۋە ھەر دەرىجىلىك يەرلىك خەلق ھۆكۈمەتلىرى مىللىي ساھەدىكى مۇھىم ئىشلارنى دوكلات قىلىش تۈزۈمىنى ساغلاملاشتۇرۇپ، مىللىي ساھەدىكى زور خەۋپ- خەتەر يوشۇرۇن خەۋپلىرىنى تەكشۈرۈش، پەرقلەندۈرۈش، باھالاش، ئاگاھلاندۇرۇش ۋە بىر تەرەپ قىلىشنى كۈچەيتىپ، خەۋپ -خەتەرنىڭ ئالدىنى ئېلىش ۋە ئۇنى بىر تەرەپ قىلىش ئىقتىدارىنى ئۆستۈرۈپ، دۆلەت بىخەتەرلىكى ۋە جەمئىيەت مۇقىملىقىنى قوغدىشى كېرەك.

**53-ماددا** ھەر دەرىجىلىك يەرلىك خەلق ھۆكۈمەتلىرى ۋە ئالاقىدار ئورگانلار زىددىيەت - ماجىرالارنىڭ ئالدىنى ئېلىش ۋە ئۇنى ھەل قىلىش ئىقتىدارى قۇرۇلۇشىنى كۈچەيتىپ، كۆپ خىل ماجىرا ھەل قىلىش مېخانىزمىنىڭ رولىنى جارى قىلدۇرۇپ، ماجىرالارنىڭ خاراكتېرىنى توغرا بېكىتىپ، قانۇن بويىچە ماجىرالارنى بىر تەرەپ قىلىشى ۋە ھەل قىلىشى، مىللەتلەر ئىتتىپاقلىقىنى قوغدىشى كېرەك. ھەرقانداق تەشكىلات ۋە شەخسنىڭ مىللىي سالاھىيەت، ئۆرپ-ئادەت، دىنىي ئېتىقاد قاتارلىقلارنى باھانە قىلىپ، قەستەن زىددىيەت پەيدا قىلىشى ياكى كەسكىنلەشتۈرۈشى، ئاممىۋى تەرتىپنى قالايمىقانلاشتۇرۇشىغا يول قويۇلمايدۇ.

**54-ماددا** پۇقرالارنىڭ مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتقا بۇزغۇنچىلىق قىلىدىغان قىلمىشلارنى ئەرز قىلىش ۋە پاش قىلىش ھوقۇقى بار؛ دۆلەت ئورگانلىرى ۋە ئۇنىڭ خىزمەتچى خادىملىرىنىڭ جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتقا ئالاقىدار خىزمەتلەردە قانۇندا بەلگىلەنگەن بۇرچىنى ئادا قىلماسلىق ياكى توغرا ئادا قىلماسلىق قىلمىشلىرىنى پاش قىلىش ھوقۇقى بار. 

بۇ قانۇن بەلگىلىمىسىگە خىلاپلىق قىلىپ، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتىغا بۇزغۇنچىلىق قىلىپ، دۆلەت مەنپەئ ەتى ياكى ئىجتىمائىي ئاممىۋى مەنپەئەتكە زىيان يەتكۈزگەنلەرگە قارىتا، خەلق تەپتىش مەھكىمىسى قانۇن بويىچە تەپتىش تەكلىپى بېرىشى ياكى ئاممىۋى مەنپەئەت دەۋاسى قىلسا بولىدۇ.

**55-ماددا** مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىيات ئۈلگە كۆرسىتىش پائالىيىتىدە، جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەلىپىنى گەۋدىلەندۈرۈش كېرەك. 

مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىيات ئىشلىرىغا گەۋدىلىك تۆھپە قوشقان كوللېكتىپ ۋە شەخسلەرگە دۆلەتنىڭ ئالاقىدار بەلگىلىمىلىرى بويىچە تەقدىرلەش مۇكاپاتى بېرىلىدۇ.

**56-ماددا** ھەر يىلى 9-ئاينىڭ تۆتىنچى ھەپتىسى مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىيات تەشۋىقات ھەپتىلىكى بولۇپ، جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش تەشۋىق- تەربىيە پائالىيەتلىرى مەركەزلىك قانات يايدۇرۇلىدۇ.

## 6-باب قانۇنىي جاۋابكارلىق

**57-ماددا** دۆلەت ئورگانلىرى ۋە ئۇنىڭ خىزمەتچى خادىملىرى جۇڭخۇا مىللەتلىرى ئورتاق گەۋدىسى ئېڭىنى مۇستەھكەملەش، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتقا ئالاقىدار قانۇندا بەلگىلەنگەن بۇرچىنى ئادا قىلماسلىق ياكى توغرا ئادا قىلماسلىق، ياكى بۇ قانۇننىڭ 58-،59-،60-،61-ماددىلىرىدا بەلگىلەنگەن قانۇنغا خىلاپ قىلمىشلارنى ۋاقتىدا توختاتماسلىق ھەمدە قانۇن بويىچە بىر تەرەپ قىلماسلىق ئەھۋالى كۆرۈلسە، ئۇنىڭ مەسئۇل تارمىقى ياكى ئالاقىدار ئورگانلار تەرىپىدىن تۈزىتىش بۇيرۇلىدۇ؛ ناچار ئاقىۋەت ياكى تەسىر پەيدا قىلغان بولسا، مەسئۇلىيەتنى ئۈستىگە ئالغان رەھبەر خادىملار ۋە بىۋاسىتە جاۋابكار خادىملارغا قانۇن بويىچە چارە كۆرۈلىدۇ؛ جىنايەت شەكىللەندۈرگەن بولسا، قانۇن بويىچە جىنايى جاۋابكارلىقى سۈرۈشتۈرۈلىدۇ.

**58-ماددا** ھەرقانداق تەشكىلات ۋە شەخس بۇ قانۇننىڭ ئالاقىدار بەلگىلىمىلىرىگە خىلاپلىق قىلىپ، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتقا بۇزغۇنچىلىق قىلسا، ناھىيە دەرىجىلىكتىن يۇقىرى مىللەت خىزمىتى، مائارىپ مەمۇرىيىتى، جامائەت خەۋپسىزلىكى، دۆلەت بىخەتەرلىكى، ئەدلىيە مەمۇرىيىتى، تۇرالغۇ ۋە شەھەر-يېزا قۇرۇلۇشى، مەدەنىيەت ۋە ساياھەت، ئاخبارات-نەشرىيات، بازار نازارەت قىلىش- باشقۇرۇش، رادىيو-تېلېۋىزىيە، تېلېگراف، تور ئۇچۇر، دىن، مەدەنىيەت يادىكارلىقلىرى قاتارلىق ئالاقىدار تارماقلار مەسئۇلىيىتى بويىچە ۋاقتىدا توختىتىدۇ، تۈزىتىشكە بۇيرۇيدۇ ھەمدە ئالاقىدار قانۇن-نىزاملارنىڭ بەلگىلىمىسى بويىچە جازا بېرىدۇ. جامائەت خەۋپسىزلىكىنى باشقۇرۇشقا خىلاپ قىلمىش شەكىللەندۈرگەن بولسا، جامائەت خەۋپسىزلىكى ئورگانلىرى قانۇن بويىچە جامائەت خەۋپسىزلىكىنى باشقۇرۇش جازاسى بېرىدۇ؛ جىنايەت شەكىللەندۈرگەن بولسا، قانۇن بويىچە جىنايى جاۋابكارلىقى سۈرۈشتۈرۈلىدۇ.

**59-ماددا** ھەرقانداق تەشكىلات ۋە شەخس مىللىي سالاھىيەتنى باھانە قىلىپ ئىشقا ئورۇنلىشىشتا كەمسىتىش، تاۋار ياكى مۇلازىمەت بىلەن تەمىنلەشنى رەت قىلىش، ياكى قانۇن-نىزاملاردا چەكلەنگەن مىللىي سالاھىيەتنى باھانە قىلىپ باشقا مۇۋاپىقسىز ئايرىم مۇئامىلە قىلىش قىلمىشلىرىنى سادىر قىلسا، ناھىيە دەرىجىلىكتىن يۇقىرى مىللەت خىزمىتى، ئادەم كۈچى بايلىقى ۋە ئىجتىمائىي كاپالەت، بازار نازارەت قىلىش -باشقۇرۇش قاتارلىق ئالاقىدار تارماقلار مەسئۇلىيىتى بويىچە تۈزىتىش كە بۇيرۇيدۇ؛ ناچار ئاقىۋەت ياكى تەسىر پەيدا قىلغان بولسا، ئاگاھلاندۇرۇش بېرى لىدۇ ياكى تەنقىد ئېلان قىلىنىدۇ. قانۇن-نىزاملاردا باشقا جازا بەلگىلىمىسى بولسا، شۇ بەلگىلىمە بويىچە بولىدۇ.

**60-ماددا** ئىجتىمائىي تەشكىلاتلار، كارخانا-كەسپىي ئورۇنلار ۋە باشقا ئىجتىمائىي تەشكىلاتلار ئۆز ئورنىدا يۈز بەرگەن مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتقا بۇزغۇنچىلىق قىلىدىغان قىلمىشلارنى ۋاقتىدا توختىتىشى كېرەك؛ ۋاقتىدا توختاتماي، ناچار ئاقىۋەت ياكى تەسىر پەيدا قىلغان بولسا، يۇقىرى دەرىجىلىك ئورگان ياكى ئالاقىدار مەسئۇل تارماقلار تەرىپىدىن ئاگاھلاندۇرۇش بېرىلىدۇ ياكى تەنقىد ئېلان قىلىنىدۇ ھەم دە بىۋاسىتە مەسئۇل باشقۇرغۇچى خادىملار ۋە باشقا بىۋاسىتە جاۋابكار خادىملارنىڭ قانۇنىي جاۋابكارلىقى ئالاقىدار قانۇن-نىزاملار بويىچە سۈرۈشتۈرۈلىدۇ.

**61-ماددا** تور تىجارەتچىلىرى بۇ قانۇن بەلگىلىمىسىگە خىلاپلىق قىلىپ، باشقۇرۇش مەسئۇلىيىتىنى ئادا قىلمىسا، تور ئۇچۇر، تېلېگراف، جامائەت خەۋپسىزلىكى، دۆلەت بىخەتەرلىكى، رادىيو- تېلېۋىزىيە قاتارلىق ئالاقىدار مەسئۇل تارماقلار تەرىپىدىن تۈزىتىشكە بۇيرۇلىدۇ، تۈزىتىشنى رەت قىلسا ياكى ئەھۋالى ئېغىر بولسا، ئالاقىدار قانۇن-نىزاملارنىڭ بەلگىلىمىسى بويىچە جازا بېرىلىدۇ.

**62-ماددا** زوراۋانلىق، تېررورلۇق پائالىيەتلىرىنى، مىللىي بۆلگۈنچىلىك پائالىيەتلىرىنى ياكى دىنىي ئاشقۇن لۇق پائالىيەتلىرىنى تەشكىللىگەن، پىلانلىغان ياكى ئېلىپ بارغان ۋە جىنايەت شەكىللەندۈرگەنلەر قانۇن بويىچە جىنايى جاۋابكارلىققا تارتىلىدۇ. 

ئالدىنقى تارماقتا كۆرسىتىلگەن ھەرىكەتلەرنى قىلىشقا قۇتراتقۇلۇق قىلغان ياكى مەبلەغ سالغان، ھەمدە جىنايەت شەكىللەندۈرگەنلەر قانۇن بويىچە جىنايى جاۋابكارلىققا تارتىلىدۇ.

**63-ماددا** جۇڭخۇا خەلق جۇمھۇرىيىتى چېگرا سىرتىدىكى تەشكىلات ۋە شەخسلەر جۇڭخۇا خەلق جۇمھۇرىيىتىگە قارىتا مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتىغا بۇزغۇنچىلىق قىلىش، مىللىي بۆلگۈنچىلىك پەيدا قىلىش قىلمىشلىرىنى سادىر قىلغانلار قانۇن بويىچە قانۇنىي جاۋابكارلىققا تارتىلىدۇ.

## 7-باب قوشۇمچە پىرىنسىپلار

**64-ماددا** ئۆلكىلىك، ئاپتونوم رايونلۇق، بىۋاسىتە قاراشلىق شەھەرلىك ۋە رايونلارغا بۆلۈنگەن شەھەرلىك، ئاپتونوم ئوبلاستلىق خەلق ۋەكىللەر قۇرۇلتايلىرى ۋە ئۇلارنىڭ دائىمىي كومىتېتلىرى يەرلىكنىڭ ئەمەلىي ئەھۋالىغا بىرلەشتۈرۈپ، مىللەتلەر ئىتتىپاقلىقى ۋە تەرەققىياتىنى ئىلگىرى سۈرىدىغان يەرلىك قانۇن-نىزاملارنى تۈزسە بولىدۇ.

**65-ماددا** بۇ قانۇن 2026-يىلى 7-ئاينىڭ 1-كۈنىدىن باشلاپ يولغا قويۇلىدۇ.
